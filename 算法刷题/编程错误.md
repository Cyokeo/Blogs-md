## 语法欠缺
- if else if
```cpp
auto check = [&nums, &n, &target](int mid) -> bool {
	bool ans = false;
	if (nums[mid] > nums[n-1])
		if (target > nums[n-1] && target < nums[mid]) ans = true;
	// 这里的else 匹配最接近的if，而不是最初的if
	else if (target > nums[n-1] || nums[mid] > target) ans = true;
	
	return ans;
};

auto check = [&nums, &n, &target](int mid) -> bool {
	bool ans = false;
	if (nums[mid] > nums[n-1])
	{
		if (target > nums[n-1] && target < nums[mid]) ans = true;
	}
	else if (target > nums[n-1] || nums[mid] > target) ans = true;
	
	return ans;
};
```

## 注意左值引用的使用 - 是否需要修改原值
- [332. 重新安排行程](https://leetcode.cn/problems/reconstruct-itinerary/)
```cpp
class Solution {
public:
    vector<string>ans;
    vector<string>path;
    bool backt(unordered_map<string, vector<string>>&umap, string& start, int n)
    {
        if (path.size() == n+1)
        {
            ans = path;
            return true;
        }
		// 这里注意使用左值引用的方式
        auto& tos = umap[start];

        bool ret = false;

        for (int i = 0; i < tos.size() && !ret; ++i)
        {
            if (tos[i] == "0") continue;
            path.push_back(tos[i]);
            tos[i] = "0";
            ret = backt(umap, path.back(), n);
            tos[i] = path.back();
            path.pop_back();
        }
        return ret;
    }
    vector<string> findItinerary(vector<vector<string>>& tickets) {
        unordered_map<string, vector<string>>umap;
        for (auto tick : tickets)
        {
            umap[tick[0]].push_back(tick[1]);
        }

        for (auto& item : umap)
        {
            sort(item.second.begin(), item.second.end(), [](const string& lc, const string& rc) -> bool {
                return lc < rc;
            });
        }
        string start = "JFK";
        path.push_back(start);
        bool ret = backt(umap, start, tickets.size());
        return ans;
    }
};
```
