# 总结

1. 主要区分普通右值引用与万能引用

## 核心概念：万能引用（Universal Reference）

```
template<typename T>
void func(T&& param);  // T&& 在这里是万能引用，不是右值引用！
```

## 示例 1：基础的值类别推导

```cpp
#include <iostream>
#include <type_traits>

template<typename T>
void test(T&& param) {
    std::cout << "=== 推导结果 ===" << std::endl;
    std::cout << "T 的类型: ";
    
    if (std::is_same<T, int>::value) std::cout << "int";
    else if (std::is_same<T, int&>::value) std::cout << "int&";
    else if (std::is_same<T, int&&>::value) std::cout << "int&&";
    
    std::cout << std::endl;
    
    std::cout << "param 的类型: ";
    if (std::is_same<decltype(param), int&>::value) std::cout << "int&";
    else if (std::is_same<decltype(param), int&&>::value) std::cout << "int&&";
    
    std::cout << std::endl << std::endl;
}

int main() {
    int x = 10;
    int& lref = x;
    int&& rref = 20;
    
    std::cout << "情况1: 传入左值变量 x" << std::endl;
    test(x);
    // T 推导为 int&
    // param 类型为 int& (int& && 折叠为 int&)
    
    std::cout << "情况2: 传入左值引用 lref" << std::endl;
    test(lref);
    // T 推导为 int&
    // param 类型为 int&
    
    std::cout << "情况3: 传入右值引用变量 rref（注意：rref 本身是左值！）" << std::endl;
    test(rref);
    // T 推导为 int&（因为 rref 作为表达式是左值）
    // param 类型为 int&
    
    std::cout << "情况4: 传入纯右值 30" << std::endl;
    test(30);
    // T 推导为 int（注意不是 int&&）
    // param 类型为 int&& (int && 就是 int&&)
    
    std::cout << "情况5: 传入 std::move(x)" << std::endl;
    test(std::move(x));
    // T 推导为 int
    // param 类型为 int&&
    
    std::cout << "情况6: 传入 std::move(rref)" << std::endl;
    test(std::move(rref));
    // T 推导为 int
    // param 类型为 int&&
    
    return 0;
}
```

## 示例 2：详细的推导过程展示

```cpp
#include <iostream>
#include <type_traits>

// 辅助函数：打印类型信息
template<typename T>
void print_type_info(const char* expr_name) {
    std::cout << "表达式: " << expr_name << std::endl;
    std::cout << "  T = ";
    
    if (std::is_same<typename std::remove_reference<T>::type, int>::value) {
        if (std::is_lvalue_reference<T>::value) 
            std::cout << "int& (左值引用)";
        else if (std::is_rvalue_reference<T>::value) 
            std::cout << "int&& (右值引用)";
        else 
            std::cout << "int (非引用)";
    }
    std::cout << std::endl;
}

template<typename T>
void func(T&& param) {
    print_type_info<T>("传入的参数");
    
    // 验证 param 的实际类型
    std::cout << "  param 类型 = ";
    if (std::is_lvalue_reference<decltype(param)>::value)
        std::cout << "左值引用";
    else if (std::is_rvalue_reference<decltype(param)>::value)
        std::cout << "右值引用";
    std::cout << std::endl << std::endl;
}

int main() {
    int x = 10;
    int&& rref = 20;
    
    // 左值表达式
    func(x);           // x 是左值
    func(rref);        // rref 虽然类型是 int&&，但作为表达式是左值！
    
    // 右值表达式
    func(10);          // 10 是纯右值
    func(x + 5);       // x + 5 是纯右值
    func(std::move(x)); // std::move(x) 是将亡值（xvalue）
    
    return 0;
}
```

## 示例 3：完美转发场景

```cpp
#include <iostream>
#include <utility>

void process(int& x) {
    std::cout << "处理左值引用: " << x << std::endl;
}

void process(int&& x) {
    std::cout << "处理右值引用: " << x << std::endl;
}

// 完美转发包装器
template<typename T>
void wrapper(T&& param) {
    std::cout << "wrapper 接收到的参数，T = ";
    
    if (std::is_lvalue_reference<T>::value)
        std::cout << "左值引用类型" << std::endl;
    else
        std::cout << "非引用类型（对应右值）" << std::endl;
    
    // 完美转发：保持原始值类别
    process(std::forward<T>(param));
    std::cout << std::endl;
}

int main() {
    int x = 42;
    int&& rref = 100;
    
    std::cout << "=== 场景1: 传递左值 ===" << std::endl;
    wrapper(x);
    // T 推导为 int&
    // forward<int&>(param) 返回 int&
    // 调用 process(int&)
    
    std::cout << "=== 场景2: 传递右值引用变量（左值表达式）===" << std::endl;
    wrapper(rref);
    // T 推导为 int&（因为 rref 是左值）
    // forward<int&>(param) 返回 int&
    // 调用 process(int&)
    
    std::cout << "=== 场景3: 传递纯右值 ===" << std::endl;
    wrapper(42);
    // T 推导为 int
    // forward<int>(param) 返回 int&&
    // 调用 process(int&&)
    
    std::cout << "=== 场景4: 传递 std::move 结果 ===" << std::endl;
    wrapper(std::move(x));
    // T 推导为 int
    // forward<int>(param) 返回 int&&
    // 调用 process(int&&)
    
    return 0;
}
```

## 示例 4：对比普通右值引用和万能引用

```cpp
#include <iostream>
#include <type_traits>

// 普通右值引用（不是模板）
void func_rvalue_only(int&& param) {
    std::cout << "普通右值引用版本" << std::endl;
}

// 万能引用（模板）
template<typename T>
void func_universal(T&& param) {
    std::cout << "万能引用版本, T = ";
    if (std::is_lvalue_reference<T>::value)
        std::cout << "左值引用";
    else
        std::cout << "非引用";
    std::cout << std::endl;
}

int main() {
    int x = 10;
    int&& rref = 20;
    
    // 普通右值引用：只能接受右值
    // func_rvalue_only(x);      // 错误！不能传左值
    // func_rvalue_only(rref);   // 错误！rref 是左值
    func_rvalue_only(30);        // 正确
    func_rvalue_only(std::move(x)); // 正确
    
    std::cout << std::endl;
    
    // 万能引用：可以接受任何值类别
    func_universal(x);           // 正确，T = int&
    func_universal(rref);        // 正确，T = int&（rref 是左值！）
    func_universal(30);          // 正确，T = int
    func_universal(std::move(x)); // 正确，T = int
    
    return 0;
}
```

## 推导规则总结表

```cpp
传入的表达式表达式值类别T 推导为T&& 折叠后param 类型

int x; func(x)左值int&int& && → int&int&
int& lref; func(lref)左值int&int& && → int&int&
int&& rref; func(rref)左值int&int& && → int&int&
func(42)纯右值intint&&int&&
func(std::move(x))将亡值intint&&int&&
```

## 关键要点

1. **模板推导看的是传入参数的值类别（左值/右值），而不是类型**
2. **即使变量类型是** `**int&&**`**，如果它有名字，作为表达式就是左值**
3. **左值推导出** `**T&**`**，右值推导出** `**T**`**（非引用）**
4. **引用折叠规则：**`**& &&**` **→** `**&**`**，其他情况保留** `**&&**`

这就是为什么 `int&& rref = 5; func(rref);` 中，`T` 会被推导为 `int&` 而不是 `int&&`！
